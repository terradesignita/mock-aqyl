# Физическая схема БД

Навигация: [[01_Readme]] | [[06_Модель_данных]] | [[12_Инфраструктура_и_DevOps]]

> **Версия:** 1.1 | **Обновлено:** 2026-04-20

## Технологический baseline

- PostgreSQL 16
- Расширения: `pgvector` (HNSW векторный индекс), `pg_trgm`, `uuid-ossp`, `pg_stat_statements`
- ORM: SQLAlchemy 2.x (DeclarativeBase, Mapped columns)
- Миграции: Alembic (или Flyway/Liquibase — один инструмент на среду)

## DDL — группа Documents

```sql
CREATE TABLE sources (
  id            VARCHAR(36) PRIMARY KEY,
  name          VARCHAR(255) NOT NULL,
  scope         TEXT NOT NULL CHECK (scope IN ('INTERNAL','EXTERNAL')),
  kind          TEXT NOT NULL CHECK (kind IN ('MANUAL','FOLDER_SYNC','ONEDRIVE')),
  root_path     VARCHAR(1024),
  active        BOOLEAN NOT NULL DEFAULT TRUE,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE documents (
  id                VARCHAR(36) PRIMARY KEY,
  source_id         VARCHAR(36) NOT NULL REFERENCES sources(id),
  title             VARCHAR(512) NOT NULL,
  author            VARCHAR(255),
  language          VARCHAR(8),
  content_type      VARCHAR(32) NOT NULL,  -- pdf/docx/pptx/mp4/wav/jpeg/...
  classification    TEXT NOT NULL CHECK (classification IN ('PUBLIC','INTERNAL','CONFIDENTIAL','RESTRICTED')),
  business_unit     VARCHAR(64) NOT NULL DEFAULT 'corporate',
  scope             TEXT NOT NULL CHECK (scope IN ('INTERNAL','EXTERNAL')),
  access_layer      VARCHAR(20) NOT NULL DEFAULT 'open',
  allowed_positions JSONB NOT NULL DEFAULT '[]',
  title_i18n        JSONB DEFAULT '{}',
  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE document_versions (
  id                       VARCHAR(36) PRIMARY KEY,
  document_id              VARCHAR(36) NOT NULL REFERENCES documents(id),
  source_path              VARCHAR(1024) NOT NULL,
  storage_uri              VARCHAR(1024) NOT NULL,
  checksum_sha256          VARCHAR(64) NOT NULL,
  processing_status        TEXT NOT NULL CHECK (processing_status IN
                             ('DISCOVERED','QUEUED','PROCESSING','INDEXED','FAILED','QUARANTINED','DUPLICATE')),
  indexed_at               TIMESTAMPTZ,
  page_count               INTEGER,
  duration_seconds         FLOAT,
  source_language          VARCHAR(8),
  extracted_text           TEXT,
  masked_text              TEXT,
  metadata_json            JSONB NOT NULL DEFAULT '{}',
  possible_duplicate       BOOLEAN NOT NULL DEFAULT FALSE,
  duplicate_of_version_id  VARCHAR(36) REFERENCES document_versions(id),
  transcription_status     VARCHAR(32),
  tagging_status           VARCHAR(32),
  ai_summary               TEXT,
  index_version            INTEGER NOT NULL DEFAULT 1,
  error_message            TEXT,
  created_at               TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_doc_version_checksum_source UNIQUE (checksum_sha256, source_path)
);

CREATE TABLE chunks (
  id                   VARCHAR(36) PRIMARY KEY,
  document_version_id  VARCHAR(36) NOT NULL REFERENCES document_versions(id),
  document_id          VARCHAR(36) NOT NULL REFERENCES documents(id),
  chunk_index          INTEGER NOT NULL,
  text                 TEXT NOT NULL,
  section_title        VARCHAR(255),
  page_from            INTEGER,
  page_to              INTEGER,
  anchor               VARCHAR(128) NOT NULL,
  token_count          INTEGER NOT NULL DEFAULT 0,
  language             VARCHAR(8),
  char_offset_start    INTEGER,
  char_offset_end      INTEGER,
  embedding_model      VARCHAR(128),
  embedding_version    INTEGER NOT NULL DEFAULT 1,
  embedding            VECTOR(768) NOT NULL,  -- pgvector
  search_vector        TSVECTOR,
  created_at           TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE transcript_segments (
  id                   VARCHAR(36) PRIMARY KEY,
  document_version_id  VARCHAR(36) NOT NULL REFERENCES document_versions(id),
  segment_index        INTEGER NOT NULL,
  start_ms             INTEGER,
  end_ms               INTEGER,
  speaker              VARCHAR(64),
  language             VARCHAR(8),
  anchor               VARCHAR(128) NOT NULL,
  text                 TEXT NOT NULL,
  created_at           TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_transcript_segment_index UNIQUE (document_version_id, segment_index)
);
```

## DDL — группа Artifacts

```sql
CREATE TABLE artifacts (
  id          VARCHAR(36) PRIMARY KEY,
  document_id VARCHAR(36) REFERENCES documents(id),
  topic_id    VARCHAR(36) REFERENCES topics(id),
  kind        VARCHAR(32) NOT NULL,    -- card / summary / podcast / flashcard / ...
  status      VARCHAR(32) NOT NULL DEFAULT 'draft',
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT ck_artifact_has_source CHECK (document_id IS NOT NULL OR topic_id IS NOT NULL)
);

CREATE TABLE artifact_versions (
  id                  VARCHAR(36) PRIMARY KEY,
  artifact_id         VARCHAR(36) NOT NULL REFERENCES artifacts(id),
  language            VARCHAR(8) NOT NULL,
  content_json        JSONB NOT NULL,
  content_text        TEXT,
  embedding           VECTOR(768),
  model_name          VARCHAR(255),
  served_language     VARCHAR(8),
  translation_pending BOOLEAN NOT NULL DEFAULT FALSE,
  prompt_version      VARCHAR(32),
  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_artifact_version_language UNIQUE (artifact_id, language)
);

CREATE TABLE artifact_citations (
  id          VARCHAR(36) PRIMARY KEY,
  version_id  VARCHAR(36) NOT NULL REFERENCES artifact_versions(id),
  chunk_id    VARCHAR(36) NOT NULL,
  quote       TEXT NOT NULL,
  anchor      VARCHAR(128),
  score       FLOAT
);
```

## DDL — группа Topics (Коллекции)

```sql
CREATE TABLE topics (
  id            VARCHAR(36) PRIMARY KEY,
  title         VARCHAR(255) NOT NULL,
  slug          VARCHAR(255) NOT NULL,
  description   TEXT,
  scope         TEXT NOT NULL CHECK (scope IN ('INTERNAL','EXTERNAL')),
  business_unit VARCHAR(64) NOT NULL DEFAULT 'corporate',
  liked         BOOLEAN NOT NULL DEFAULT FALSE,
  pinned        BOOLEAN NOT NULL DEFAULT FALSE,
  difficulty    VARCHAR(32) NOT NULL DEFAULT 'foundation',
  image_url     VARCHAR(1024),
  parent_id     VARCHAR(36) REFERENCES topics(id),
  owner_email   VARCHAR(255),
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_topics_slug UNIQUE (slug)
);

CREATE TABLE topic_members (
  id          VARCHAR(36) PRIMARY KEY,
  topic_id    VARCHAR(36) NOT NULL REFERENCES topics(id) ON DELETE CASCADE,
  user_email  VARCHAR(255) NOT NULL,
  role        VARCHAR(20) NOT NULL,   -- owner / editor / viewer
  added_by    VARCHAR(255) NOT NULL,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_topic_members_link UNIQUE (topic_id, user_email)
);

CREATE TABLE topic_documents (
  id          VARCHAR(36) PRIMARY KEY,
  topic_id    VARCHAR(36) NOT NULL REFERENCES topics(id),
  document_id VARCHAR(36) NOT NULL REFERENCES documents(id),
  is_primary  BOOLEAN NOT NULL DEFAULT FALSE,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_topic_documents_link UNIQUE (topic_id, document_id)
);

CREATE TABLE notifications (
  id          SERIAL PRIMARY KEY,
  user_email  VARCHAR(256) NOT NULL,
  type        VARCHAR(32) NOT NULL,
  topic_id    VARCHAR(36) REFERENCES topics(id) ON DELETE SET NULL,
  actor_email VARCHAR(256) NOT NULL,
  payload     TEXT,
  is_read     BOOLEAN NOT NULL DEFAULT FALSE,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
```

## Индексация и производительность

### Векторный индекс (pgvector HNSW)
```sql
CREATE INDEX idx_chunks_embedding ON chunks
  USING hnsw (embedding vector_cosine_ops)
  WITH (m = 16, ef_construction = 64);
```

### Полнотекстовый поиск (GIN)
```sql
CREATE INDEX idx_chunks_search ON chunks USING GIN (search_vector);
```

### Составные B-tree индексы
```sql
-- document_versions
CREATE INDEX idx_document_versions_document ON document_versions (document_id, created_at);

-- chunks
CREATE INDEX idx_chunks_document_version ON chunks (document_version_id, chunk_index);

-- transcript_segments
CREATE INDEX idx_transcript_segments_version ON transcript_segments (document_version_id, segment_index);

-- topics
CREATE INDEX idx_topics_scope_updated ON topics (scope, updated_at);
CREATE INDEX idx_topics_parent ON topics (parent_id);

-- topic_members
CREATE INDEX idx_topic_members_user_email ON topic_members (user_email);

-- topic_documents
CREATE INDEX idx_topic_documents_topic ON topic_documents (topic_id, created_at);
CREATE INDEX idx_topic_documents_document ON topic_documents (document_id);

-- artifacts
CREATE INDEX idx_artifacts_document ON artifacts (document_id, created_at);

-- artifact_versions
CREATE INDEX idx_artifact_versions_artifact ON artifact_versions (artifact_id, created_at);

-- notifications
CREATE INDEX idx_notifications_user_read_created ON notifications (user_email, is_read, created_at);
CREATE INDEX idx_notifications_topic ON notifications (topic_id);

-- generated_assets
CREATE INDEX idx_generated_assets_owner ON generated_assets (owner_type, owner_id, created_at);
CREATE INDEX idx_generated_assets_kind ON generated_assets (kind, status);

-- pipeline_jobs
CREATE INDEX idx_pipeline_jobs_resource ON pipeline_jobs (resource_type, resource_id, created_at);
CREATE INDEX idx_pipeline_jobs_status ON pipeline_jobs (status, created_at);

-- query_logs
CREATE INDEX idx_query_logs_user_time ON query_logs (user_email, created_at);

-- tags
CREATE INDEX idx_topic_tags_topic ON topic_tags (topic_id);
CREATE INDEX idx_topic_tags_tag ON topic_tags (tag_id);
CREATE INDEX idx_document_tags_document ON document_tags (document_id);
CREATE INDEX idx_document_tags_tag ON document_tags (tag_id);
```

## Партиционирование

- `query_logs` — партиционировать по месяцу (`created_at`). Ретеншн логов: 180 дней (по комплаенсу может быть выше).
- `pipeline_jobs` — при высокой нагрузке рекомендуется партиционирование по статусу или месяцу.

## Миграции

- Forward-only migrations.
- Rollback через compensating migration.
- Каждая миграция содержит:
  - pre-check;
  - migration DDL;
  - post-check.

## Политика изменений схемы

- Breaking schema changes только в major release.
- Любое удаление колонки — через 2 фазы:
  1. deprecate (колонка помечается как deprecated, код перестаёт её читать)
  2. remove после релиза и проверки consumers.
- Добавление колонки — всегда `NULLABLE` или с `DEFAULT`, чтобы не блокировать таблицу.
